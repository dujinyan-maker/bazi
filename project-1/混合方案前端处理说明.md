# 混合方案前端处理说明

## 方案概述

**混合方案**：连接时推送完整列表 + 发送时广播新消息

### 工作流程

1. **客户端连接 WebSocket**
2. **服务器推送完整列表**：最新的 50 条祝福语（`type: "blessing_list"`）
3. **服务器推送连接成功消息**：`type: "connected"`
4. **有新的祝福语时**：服务器广播单条新消息（`type: "new_blessing"`）

---

## 前端代码实现

### JavaScript 原生示例

```javascript
const ws = new WebSocket('wss://192.168.1.10:8081/ws/blessing');
const messages = []; // 存储祝福语列表

ws.onopen = () => {
    console.log('✅ WebSocket 连接成功！');
};

ws.onmessage = (event) => {
    const message = JSON.parse(event.data);
    console.log('收到消息:', message);
    
    // 处理不同类型的消息
    switch (message.type) {
        case 'connected':
            // 连接成功消息
            console.log('连接成功:', message.message);
            break;
            
        case 'blessing_list':
            // 收到完整的祝福语列表（连接时推送）
            const blessingList = message.data; // List<BlessingMessageResponse>
            console.log('收到祝福语列表，数量:', blessingList.length);
            
            // 清空旧数据，加载新列表
            messages.length = 0;
            blessingList.forEach(msg => {
                messages.push(msg);
            });
            
            // 更新 UI
            updateUI(messages);
            break;
            
        case 'new_blessing':
            // 收到新的祝福语（发送时广播）
            const newBlessing = message.data; // BlessingMessageResponse
            console.log('收到新祝福语:', newBlessing);
            
            // 添加到列表顶部
            messages.unshift(newBlessing);
            
            // 限制列表长度（可选）
            if (messages.length > 100) {
                messages.pop(); // 移除最旧的
            }
            
            // 更新 UI
            addNewBlessingToUI(newBlessing);
            break;
            
        case 'pong':
            // 心跳响应
            console.log('收到心跳响应');
            break;
            
        default:
            console.warn('未知的消息类型:', message.type);
    }
};

// 更新 UI（显示完整列表）
function updateUI(msgs) {
    const container = document.getElementById('messages');
    container.innerHTML = ''; // 清空
    
    msgs.forEach(msg => {
        const item = createBlessingItem(msg);
        container.appendChild(item);
    });
}

// 添加新祝福语到 UI
function addNewBlessingToUI(blessing) {
    const container = document.getElementById('messages');
    const item = createBlessingItem(blessing);
    container.insertBefore(item, container.firstChild);
    
    // 限制显示数量
    while (container.children.length > 100) {
        container.removeChild(container.lastChild);
    }
}

// 创建祝福语元素
function createBlessingItem(blessing) {
    const div = document.createElement('div');
    div.className = 'blessing-item';
    div.id = 'msg-' + blessing.id;
    
    div.innerHTML = `
        <span class="nickname">${blessing.nickname}:</span>
        <span class="content">${blessing.content}</span>
        <span class="time">${formatTime(blessing.createdTime)}</span>
    `;
    
    return div;
}

function formatTime(timeStr) {
    const date = new Date(timeStr);
    return date.toLocaleString('zh-CN');
}
```

### Vue.js 示例

```vue
<template>
  <div class="blessing-container">
    <div class="status" :class="{ connected: wsConnected }">
      {{ wsStatus }}
    </div>
    
    <div class="messages">
      <div 
        v-for="message in messages" 
        :key="message.id" 
        class="blessing-item"
      >
        <span class="nickname">{{ message.nickname }}:</span>
        <span class="content">{{ message.content }}</span>
        <span class="time">{{ formatTime(message.createdTime) }}</span>
      </div>
    </div>
  </div>
</template>

<script>
export default {
  name: 'BlessingList',
  data() {
    return {
      ws: null,
      messages: [],
      wsConnected: false,
      wsStatus: '连接中...',
      wsUrl: 'wss://192.168.1.10:8081/ws/blessing'
    };
  },
  mounted() {
    this.connectWebSocket();
  },
  beforeUnmount() {
    this.disconnectWebSocket();
  },
  methods: {
    connectWebSocket() {
      this.ws = new WebSocket(this.wsUrl);
      
      this.ws.onopen = () => {
        console.log('WebSocket 连接已建立');
        this.wsConnected = true;
        this.wsStatus = '已连接';
      };
      
      this.ws.onmessage = (event) => {
        const message = JSON.parse(event.data);
        
        switch (message.type) {
          case 'connected':
            console.log('连接成功:', message.message);
            break;
            
          case 'blessing_list':
            // 收到完整列表，替换当前列表
            this.messages = message.data || [];
            console.log('收到祝福语列表，数量:', this.messages.length);
            break;
            
          case 'new_blessing':
            // 收到新祝福语，添加到列表顶部
            const newBlessing = message.data;
            this.messages.unshift(newBlessing);
            
            // 限制列表长度
            if (this.messages.length > 100) {
              this.messages = this.messages.slice(0, 100);
            }
            console.log('收到新祝福语:', newBlessing);
            break;
            
          case 'pong':
            console.log('收到心跳响应');
            break;
        }
      };
      
      this.ws.onclose = () => {
        console.log('WebSocket 连接已关闭');
        this.wsConnected = false;
        this.wsStatus = '连接已断开';
      };
      
      this.ws.onerror = (error) => {
        console.error('WebSocket 错误:', error);
        this.wsStatus = '连接错误';
      };
    },
    
    disconnectWebSocket() {
      if (this.ws) {
        this.ws.close();
        this.ws = null;
      }
    },
    
    formatTime(timeStr) {
      const date = new Date(timeStr);
      return date.toLocaleString('zh-CN');
    }
  }
};
</script>
```

```jsx
import React, { useState, useEffect, useRef } from 'react';

function BlessingList() {

  const [wsConnected, setWsConnected] = useState(false);

  const wsRef = useRef(null);
  
  const WS_URL = 'wss://192.168.1.10:8081/ws/blessing';

  useEffect(() => {
    const ws = new WebSocket(WS_URL);
    wsRef.current = ws;

    ws.onopen = () => {
      console.log('WebSocket 连接已建立');
      setWsConnected(true);
      setWsStatus('已连接');

    ws.onmessage = (event) => {
      const message = JSON.parse(event.data);
      
      switch (message.type) {
        case 'connected':
          console.log('连接成功:', message.message);
          break;
          
        case 'blessing_list':
          // 收到完整列表，替换当前列表
          setMessages(message.data || []);

          break;
          
        case 'new_blessing':
          // 收到新祝福语，添加到列表顶部
          setMessages(prevMessages => {
            const newMessages = [message.data, ...prevMessages];

          });
          console.log('收到新祝福语:', message.data);
          break;
          
        case 'pong':
          console.log('收到心跳响应');
          break;
          
        default:
          console.warn('未知消息类型:', message.type);
      }
    };

    ws.onclose = () => {
      console.log('WebSocket 连接已关闭');
      setWsConnected(false);
      setWsStatus('连接已断开');
    };

    ws.onerror = (error) => {
      console.error('WebSocket 错误:', error);
      setWsStatus('连接错误');
    };

    return () => {
      if (ws) {
        ws.close();
      }
    };
  }, []);

  const formatTime = (timeStr) => {
    const date = new Date(timeStr);
    return date.toLocaleString('zh-CN');
  };

  return (

        padding: '10px',
        margin: '10px 0',
        borderRadius: '5px',
        background: wsConnected ? '#d4edda' : '#f8d7da',
        color: wsConnected ? '#155724' : '#721c24'
      }}>
        {wsStatus}
      </div>
      
      <div>
        {messages.map(message => (
          <div key={message.id} style={{
            padding: '10px',
            margin: '10px 0',
            background: '#f5f5f5',
            borderRadius: '5px'

            <span style={{ color: '#999', fontSize: '12px', float: 'right' }}>
              {formatTime(message.createdTime)}
            </span>
          </div>
        ))}
      </div>
    </div>
  );
}
```

## 消息类型说明

### 1. 连接成功消息
```json
{
  "type": "connected",
  "timestamp": 1704614400000,
  "data": {
    "message": "连接成功"
  }
}
```

### 2. 祝福语列表消息（连接时推送）
```json
{
  "type": "blessing_list",
  "timestamp": 1704614400000,
  "data": [
    {
      "id": 100,
      "userId": 1,
      "nickname": "用户8000",
      "content": "祝大家新年快乐！",
      "createdTime": "2026-01-08T10:00:00"
    },
    {
      "id": 99,
      "userId": 2,
      "nickname": "用户8001",
      "content": "万事如意！",
      "createdTime": "2026-01-08T09:59:00"

    // ... 最多50条
  ]
}
```

**前端处理**：
- 清空当前列表
- 用新列表替换
- 更新 UI

### 3. 新祝福语消息（发送时广播）
```json
{
  "type": "new_blessing",
  "timestamp": 1704614400000,
  "data": {
    "id": 101,
    "userId": 3,
    "nickname": "用户8002",
    "content": "身体健康！",
    "createdTime": "2026-01-08T10:01:00"
```

**前端处理**：
- 添加到列表顶部
- 动画显示新消息
- 限制列表长度

```json
{
  "type": "pong",

  }
```

## 处理流程

### 连接流程
```
1. 客户端连接 WebSocket
   ↓
2. 服务器推送 connected 消息
   ↓
3. 服务器推送 blessing_list 消息（完整列表）
   ↓
4. 客户端显示完整列表
```

### 新消息流程
```
1. 用户A发送祝福语
   ↓
2. 服务器保存到数据库
   ↓
3. 服务器广播 new_blessing 消息
   ↓
4. 所有连接的客户端收到新消息
   ↓
5. 客户端添加到列表顶部并显示
```

---

## 注意事项

1. **消息顺序**：
   - 连接时可能先收到 `connected`，再收到 `blessing_list`
   - 客户端应该按顺序处理消息

2. **去重处理**：
   - `blessing_list` 和 `new_blessing` 中可能包含相同的消息ID
   - 建议根据 ID 去重，避免重复显示

3. **列表长度限制**：
   - 服务器推送最多 50 条（可配置）
   - 客户端可以限制显示更多（如 100 条）

4. **断线重连**：
   - 重连后会再次收到完整列表
   - 需要清空旧数据，避免重复

5. **去重优化示例**：
```javascript
// 使用 Map 根据 ID 去重
const messageMap = new Map();

// 处理列表
if (message.type === 'blessing_list') {
    messageMap.clear();
    message.data.forEach(msg => {
        messageMap.set(msg.id, msg);
    });
    messages = Array.from(messageMap.values());
}

// 处理新消息
if (message.type === 'new_blessing') {
    if (!messageMap.has(message.data.id)) {
        messageMap.set(message.data.id, message.data);
        messages.unshift(message.data);
    }
}
```
